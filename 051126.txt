2) Builtin execution with redirections: always restore stdin/stdout

In exec.c, replace run_builtin_cmd with:

    int run_builtin_cmd(t_data *data, t_cmd *cmd)
    {
        int in_fd;
        int out_fd;
        int ret;

        in_fd = dup(STDIN_FILENO);
        out_fd = dup(STDOUT_FILENO);
        if (in_fd < 0 || out_fd < 0)
        {
            if (in_fd >= 0)
                close(in_fd);
            if (out_fd >= 0)
                close(out_fd);
            perror("dup");
            data->status = ERROR;
            return (ERROR);
        }

        if ((cmd->infile && redirect_input(cmd) == -1)
            || (cmd->outfile && redirect_output(cmd) == -1))
        {
            dup2(in_fd, STDIN_FILENO);
            dup2(out_fd, STDOUT_FILENO);
            close(in_fd);
            close(out_fd);
            data->status = ERROR;
            return (ERROR);
        }

        ret = execute_builtin(data, cmd);

        dup2(in_fd, STDIN_FILENO);
        dup2(out_fd, STDOUT_FILENO);
        close(in_fd);
        close(out_fd);

        data->status = ret;
        return (ret);
    }

Why this matters:
- Failed redirection should not leave shell stdout/stdin broken.
- You avoid fd leaks and prompt corruption.
